OPTIMIZATION 3: Start stale lease reaper (3 min)
-----------------------------------------------
Create file: stale_lease_reaper.py

#!/usr/bin/env python3
import os, time, psycopg2
from datetime import datetime

while True:
    try:
        conn = psycopg2.connect(os.environ['DATABASE_URL'])
        cur = conn.cursor()
        cur.execute("""
            UPDATE harvest_jobs 
            SET status='pending', lease_owner=NULL 
            WHERE status='leased' 
            AND leased_at < NOW() - INTERVAL '3 minutes'
        """)
        reclaimed = cur.rowcount
        conn.commit()
        if reclaimed > 0:
            print(f"[{datetime.now().strftime('%H:%M:%S')}] Reclaimed {reclaimed} stale leases")
        cur.close()
        conn.close()
    except Exception as e:
        print(f"Error: {e}")
    time.sleep(60)

Add these lines:
from urllib.parse import urlparse
from collections import defaultdict
from threading import Lock

host_semaphores = {}
host_lock = Lock()

def get_host_semaphore(url, max_per_host=6):
    """Limit concurrent requests per host"""
    host = urlparse(url).netloc
    with host_lock:
        if host not in host_semaphores:
            from threading import Semaphore
            host_semaphores[host] = Semaphore(max_per_host)
        return host_semaphores[host]

def batch_insert_images(conn, images_batch):
    """Insert multiple images in one transaction"""
    if not images_batch:
        return 0
    
    cur = conn.cursor()
    try:
        from psycopg2.extras import execute_values
        sql = """
            INSERT INTO orchid_images 
            (taxonomy_id, image_url, source, image_type, country, latitude, 
             longitude, observation_date, year, occurrence_metadata, created_at)
            VALUES %s
            ON CONFLICT (image_url) DO NOTHING
        """
        
        values = []
        for img in images_batch:
            values.append((
                img['taxonomy_id'],
                img['url'],
                img['source'],
                img.get('type', 'observation'),
                img.get('country'),
                img.get('lat'),
                img.get('lon'),
                img.get('date'),
                img.get('year'),
                json.dumps(img.get('occurrence_metadata', {})),
                'NOW()'
            ))
        
        execute_values(cur, sql, values)
        inserted = cur.rowcount
        conn.commit()
        return inserted
    except Exception as e:
        conn.rollback()
        return 0
    finally:
        cur.close()
