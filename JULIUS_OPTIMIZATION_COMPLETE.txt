    lease_owner,
    COUNT(*) as jobs_completed
FROM harvest_jobs
WHERE status = 'completed'
    AND completed_at > NOW() - INTERVAL '1 hour'
GROUP BY lease_owner
ORDER BY COUNT(*) DESC;

    COUNT(*) as total_images_last_hour,
