EXAMPLE API CALL (JavaScript):

// Example: Generate culture sheet
async function generateCultureSheet(speciesId, location) {
  const response = await fetch('https://orchid-continuum-online.onrender.com/api/culture-sheets/generate', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      taxonomy_id: speciesId,
      latitude: location.lat,
      longitude: location.lng,
      city: location.city,
      country: location.country,
      sections: ['temperature', 'light', 'water', 'humidity', 'potting', 'fertilizer'],
      artwork_style: 'artistic'
    })
  });
  
  const data = await response.json();
  displayCultureSheet(data);
}
